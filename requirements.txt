# Asana — official v5 SDK. The package name is "asana"; "python-asana" on PyPI is
# unrelated and abandoned. v5 is a complete rewrite of v3 (Configuration + ApiClient
# + per-resource API classes); v3-era tutorials do not apply.
asana>=5.2,<6

# Data ingestion. The sample Tableau export is UTF-16 LE + tab-delimited
# (despite the .csv extension), and Tableau occasionally produces .xlsx instead —
# pandas + openpyxl covers both.
pandas>=2.2
openpyxl>=3.1

# Fuzzy vendor matching (Asana task names vs Tableau Vendor formatting variants).
rapidfuzz>=3.9

# HTTP — used by the Asana SDK + ad-hoc requests calls.
requests>=2.32

# Use the OS trust store for TLS verification instead of certifi's bundle.
# Critical on corporate networks (life.church has SSL inspection that re-signs
# outbound traffic with a corp CA installed in the Windows machine cert store
# but NOT in certifi). engine.main calls truststore.inject_into_ssl() at the
# very top of the module; the try/except in that block keeps things working
# in dev venvs where this dep is absent.
truststore>=0.10

# Local-run convenience: .env loaded automatically by engine.main when present.
# CI does not use this — secrets come from GitHub Actions environment vars directly.
python-dotenv>=1.0

# Local web UI (engine.ui). Operator launches via `EngineApp.exe --ui` and edits
# Needs Tagging / browses Dashboard / inspects Run Log in the browser. Bound to
# localhost:8080 only; no auth (single-user single-machine context). Pico.css is
# served from a CDN so there's no static asset pipeline to manage.
flask>=3.0

pytest>=8.2
